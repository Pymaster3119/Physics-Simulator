#version 330 core

const vec3[6] positions = vec3[](
    vec3(-1, -1, 0),
    vec3(-1, 1, 0),
    vec3(1, 1, 0),
    vec3(1, 1, 0),
    vec3(-1, -1, 0),
    vec3(1, -1, 0)
);

const vec3[6] colors = vec3[](
    vec3(1, 1, 1),
    vec3(1, 1, 1),
    vec3(1, 1, 1),
    vec3(1, 1, 1),
    vec3(1, 1, 1),
    vec3(1, 1, 1)
);

layout (location = 0) in vec3 vertex_position;

out vec3 fragment_color;
out vec3 vertex_positiont;
void main() {
    gl_Position = vec4(positions[gl_VertexID], 1.0);
    fragment_color = colors[gl_VertexID];
    vertex_positiont = vertex_position;
}
