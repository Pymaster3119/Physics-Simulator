#version 330 core

in vec3 fragment_color;
in vec3 vertex_positiont;
out vec4 screen_color;

uniform float u_screenWidth;
uniform float u_screenHeight;

uniform vec3 u_cameraPos;
uniform vec3 u_cameraAngle;

uniform mat3 u_rotation_matrix;

uniform int u_pointCount;
uniform samplerBuffer u_points_tex;
uniform samplerBuffer u_star_velocities;

uniform int u_trakedStar;

float customDot (vec3 a, vec3 b)
{
    //TODO: update
    return dot(a, b);
}

vec3 customNormalize (vec3 a)
{
    //TODO: update
    return normalize(a);
}


float getRadius(int i) {
    vec4 t = texelFetch(u_points_tex, i);
    return t.a;
}

float getGamma(int i) {
    vec4 t = texelFetch(u_star_velocities, i);
    return t.a;
}

vec3 getVhat(int i) {
    vec4 t = texelFetch(u_star_velocities, i);
    return t.xyz;
}

vec3 getPoint(int i) {
    vec4 t = texelFetch(u_points_tex, i);
    vec3 pos = t.xyz;
    float gamma = getGamma(i);
    vec3 vhat = getVhat(i);

    vec3 dparallel = customDot(pos, vhat) * vhat;
    vec3 dperp = pos - dparallel;
    pos = dparallel / gamma + dperp;
    return pos;
}

vec3 calculate_intersection(vec3 worldRay, vec3 point, float radius, int i) {
    //Transform to compressed space
    vec3 vhat = getVhat(i);
    float gamma = getGamma(i);
    vec3 dparallel = customDot(worldRay, vhat) * vhat;
    vec3 dperpendicular = worldRay - dparallel;
    worldRay = gamma * dparallel + dperpendicular;

    vec3 oc = - point;
    vec3 ocparallel = customDot(oc, vhat) * vhat;
    vec3 ocperp = oc - ocparallel;
    oc = gamma * ocparallel + ocperp;

    //intersect
    float a = customDot(worldRay, worldRay);
    float b = 2 * customDot(oc, worldRay);
    float c = customDot(oc, oc) - radius * radius;

    float discriminant = b*b - 4*a*c;
    if (discriminant < 0) {return vec3(9999999, 0,0);} //no hit :(

    float sqrt_disc = sqrt(discriminant);
    float t1 = (-b - sqrt_disc) / (2*a);
    float t2 = (-b + sqrt_disc) / (2*a);
    float unmodified_dist = -1000000;
    if (t1 >= 0) {unmodified_dist = t1;}
    if (t2 >= 0 && t2 < unmodified_dist) {unmodified_dist = t2;}

    if (unmodified_dist < -1) {return vec3(9999999, 0,0);} //no hit :(

    vec3 intersection_point = oc + unmodified_dist * worldRay;
    
    //Untransform
    vec3 intparallel = customDot(intersection_point, vhat) * vhat;
    vec3 intperpendicular = intersection_point - intparallel;
    vec3 untransformed = intparallel / gamma + intperpendicular;
    return point + untransformed;
}


void main() {
    //Convert vertical FOV to radians
    float fovRadians = radians(60);
    float tanHalfFov = tan(fovRadians / 2.0);
    
    //Get position values normalized between -1 and 1 accounting for aspect ratio
    vec2 pixelPos = gl_FragCoord.xy;
    pixelPos[0] = pixelPos[0] / u_screenWidth;
    pixelPos[1] = pixelPos[1] / u_screenHeight;
    pixelPos = pixelPos * 2 - 1;
    pixelPos[0] = pixelPos[0] * (u_screenWidth / u_screenHeight);
    pixelPos = pixelPos * tanHalfFov;

    //create the local ray and transform it into world ray
    vec3 localRay = vec3(pixelPos[0], pixelPos[1], 1);
    vec3 worldRay = u_rotation_matrix * localRay;
    worldRay = customNormalize(worldRay);

    //Calculate intersection with points
    bool intersected = true;
    float closest_dist = 999999;
    int closest_index = -1;
    vec3 closest_center = getPoint(0);
    vec3 closest_intersection = getPoint(0);
    for (int i = 0; i < u_pointCount; i++) {
        vec3 inter = calculate_intersection(worldRay, getPoint(i), getRadius(i), i);
        float dist = length(inter - u_cameraPos);
        if (closest_dist > dist) {closest_dist = dist; closest_center = getPoint(i); closest_index = i; closest_intersection = inter;}
    }
    if (closest_dist < 999999){
        vec3 vhat = getVhat(closest_index);
        float gamma = getGamma(closest_index);
        vec3 r = closest_intersection - closest_center;
        vec3 rparallel = customDot(r, vhat) * vhat;
        vec3 rperp = r - rparallel;
        vec3 normal = rperp + rparallel / (gamma * gamma);
        normal = customNormalize(normal);
        
        vec3 lightDir = customNormalize(vec3(1.0, 1.0, -1.0));
        float color = max(0.0, customDot(normal, lightDir)) + 0.1;
        
        if (u_trakedStar != closest_index) { screen_color = vec4(color,color,color,1);}
        else { screen_color = vec4(color,0,0,1);}
    } else {screen_color = vec4(0, 0, 0, 1);}
}
